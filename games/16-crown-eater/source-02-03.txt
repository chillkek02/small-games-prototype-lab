 const acts=Game.loadout.map((m,i)=>({label:`${moveName(m)}\n→ ${moveName(moveId)}`,fn:()=>{Game.loadout[i]=moveId;Game.selectedMove=moveId;toast(I18n.t('mutationEquipped',{move:moveName(moveId)}));resume?.()}}));showModal(I18n.t('replaceTitle'),I18n.t('replaceText',{move:moveName(moveId)}),acts)}

function playerHit(amount,source){let dmg=amount;if(Game.upgrades.has('hardShell')&&!Game.hardShellUsed&&dmg>0){Game.hardShellUsed=true;dmg=Math.max(0,dmg-1);toast(I18n.t('hardShell'),800)}if(dmg>0&&Game.player.shield>0){const b=Math.min(Game.player.shield,dmg);Game.player.shield-=b;dmg-=b}if(dmg<=0)return;Game.player.hp-=dmg;Game.runDamage+=dmg;records.damageTaken+=dmg;saveRecords();Audio.hit();FX.hitShake(9,.22);FX.addFlash('#e0525f',.17);const [cx,cy]=cellCenter(Game.player.x,Game.player.y);FX.float(cx,cy,`-${dmg} ${I18n.t('hp')}`,'#ff7780',23);toast(`${enemyName(source.type)} → -${dmg} ${I18n.t('hp')}`,850)}
function promotePawn(e){Game.promotionsThisEncounter++;const forms=['knight','bishop','rook'],type=forms[(e.id+Game.encounter+Game.turn)%forms.length];e.type=type;e.guard=GUARD_BASE[type]||1;if(type==='rook'&&Game.burdens.has('rookArmor'))e.guard++;e.maxGuard=e.guard;e.clockMax=actionClockMax(type);e.clock=e.clockMax;e.intent='attack';e.justPromoted=true;Audio.unlockS();FX.burst(...cellCenter(e.x,e.y),24,'#f3c75d',150);toast(`${I18n.t('promoted')} → ${enemyName(type)}`,1100)}

function processCollapse(){if(!Game.collapsePlan.length)return true;for(const c of Game.collapsePlan){if(c.done||c.round>Game.turn)continue;const k=key(c.x,c.y),occ=enemyAt(c.x,c.y);if(Game.player.x===c.x&&Game.player.y===c.y){playerHit(2,{type:'rook'});FX.float(...cellCenter(c.x,c.y),I18n.t('collapseWarning'),'#ff9a83',14);if(Game.player.hp<=0)return false;continue}if(occ){if(occ.guard>0)occ.guard=0;killEnemy(occ,'hazard')}Game.hazards.set(k,'abyss');c.done=true;FX.burst(...cellCenter(c.x,c.y),18,'#6f6470',120)}return true}

function firstCourtTarget(e){for(const p of e.telegraph){if(Game.player.x===p.x&&Game.player.y===p.y)return {kind:'player',target:Game.player};if(hazardAt(p.x,p.y)==='mine')return {kind:'mine',x:p.x,y:p.y};const t=enemyAt(p.x,p.y,e.id);if(t)return {kind:'enemy',target:t}}return null}

 if(Game.screen!=='play')return;Game.chainMode=false;Game.chainRemaining=0;Game.selectedDest=null;Game.selectedAction='move';const active=Game.enemies.filter(e=>e.clock<=1);
 for(const e of [...active]){if(!Game.enemies.includes(e))continue;if(e.intent==='promote'){promotePawn(e);continue}if(e.staggered)continue;if(['attack','charge','hunt','curse'].includes(e.intent)){e.telegraph=threatFor(e);const hit=firstCourtTarget(e);if(hit?.kind==='player'){playerHit(e.damage,e);if(Game.player.hp<=0){defeat();return}}else if(hit?.kind==='enemy'){applyDamageEnemy(hit.target,e.damage,'friendly');Audio.push();FX.float(...cellCenter(hit.target.x,hit.target.y),I18n.t('crossfire'),'#ffcf78',18)}else if(hit?.kind==='mine'){detonateMine(hit.x,hit.y,'court');if(Game.screen==='defeat')return}}}
 for(const e of [...active])if(Game.enemies.includes(e)&&e.type==='blood_bishop'&&e.specialTiles.length){const p=e.specialTiles[0];Game.hazards.set(key(p.x,p.y),'cursed')}if(hazardAt(Game.player.x,Game.player.y)==='cursed'){playerHit(1,{type:'blood_bishop'});if(Game.player.hp<=0){defeat();return}}
 const king=active.find(e=>Game.enemies.includes(e)&&e.type==='king');if(king){if(king.intent==='buff'){Game.royalBuff++;for(const e of Game.enemies)if(e.royalGuard){e.clock=Math.max(1,e.clock-1);e.guard=Math.min(e.maxGuard+1,e.guard+1)}toast('ROYAL GUARDS STRENGTHENED',900)}if(king.intent==='summon'){summonPawnNearKing(king)}}
 for(const e of Game.enemies){if(e.staggered){e.staggered=false;e.clock=Math.max(1,e.clockMax);continue}e.clock--;if(e.clock<=0)e.clock=e.clockMax}
 if(!processReinforcements())return;if(!processCollapse())return;Game.roundsSurvived++;Game.turn++;Game.actionsLeft=Game.maxActions;Game.mawReady=true;Game.momentumUsed=false;for(const k of Object.keys(Game.cooldowns))Game.cooldowns[k]=Math.max(0,Game.cooldowns[k]-1);prepareTurn(true);captureRoundSnapshot();checkEncounterClear();
